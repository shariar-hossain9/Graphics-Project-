void rect(float x1, float y1, float x2, float y2,
          float r, float g, float b)
{
    glColor3f(r, g, b);

    glBegin(GL_QUADS);

    glVertex2f(x1, y1);
    glVertex2f(x2, y1);
    glVertex2f(x2, y2);
    glVertex2f(x1, y2);

    glEnd();
}


void circle(float x, float y, float radius,
            float red, float green, float blue)
{
    glColor3f(red, green, blue);

    glBegin(GL_POLYGON);

    for (int i = 0; i < 100; i++)
    {
        float angle = 2 * 3.1416 * i / 100;

        glVertex2f(
            x + radius * cos(angle),
            y + radius * sin(angle)
        );
    }

    glEnd();
}


void text(float x, float y, const char str[],
          float r, float g, float b)
{
    glColor3f(r, g, b);
    glRasterPos2f(x, y);

    for (int i = 0; str[i] != '\0'; i++)
    {
        glutBitmapCharacter(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }
}

void centeredText(float centerX, float y,
                  const char str[],
                  float r, float g, float b)
{
    int pixelWidth = 0;

    // Calculate total text width in pixels
    for (int i = 0; str[i] != '\0'; i++)
    {
        pixelWidth += glutBitmapWidth(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }

    // Convert pixel width into OpenGL coordinates
    float windowWidth = glutGet(GLUT_WINDOW_WIDTH);
    float worldWidth = pixelWidth * 200.0f / windowWidth;

    // Draw the text from its calculated starting position
    text(
        centerX - worldWidth / 2.0f,
        y,
        str,
        r, g, b
    );
}

bool checkVerticalCollision(float front, float back)
{
    if (fabs(front - back) < safeDistance)
    {
        return true;
    }

    return false;
}


bool checkHorizontalCollision(float front, float back)
{
    if (fabs(front - back) < safeDistance)
    {
        return true;
    }

    return false;
}

void moveNegativeCar(float &position,
                     float stopPosition,
                     bool greenSignal)
{
    /*
        Used by:
        North-to-South cars
        East-to-West cars
    */

    // Green signal or car already crossed the stopping point
    if (greenSignal || position < stopPosition)
    {
        position -= speedValue;
    }
    else if (position > stopPosition)
    {
        // Approach the zebra crossing
        position -= speedValue;

        // Prevent the car from crossing during red/yellow
        if (position < stopPosition)
        {
            position = stopPosition;
        }
    }
}


void movePositiveCar(float &position,
                     float stopPosition,
                     bool greenSignal)
{
    /*
        Used by:
        South-to-North cars
        West-to-East cars
    */

    // Green signal or car already crossed the stopping point
    if (greenSignal || position > stopPosition)
    {
        position += speedValue;
    }
    else if (position < stopPosition)
    {
        // Approach the zebra crossing
        position += speedValue;

        // Prevent the car from crossing during red/yellow
        if (position > stopPosition)
        {
            position = stopPosition;
        }
    }
}

void vehicles()
{
    /*
        Signal states:

        signal 0 = vertical green
        signal 1 = vertical yellow
        signal 2 = horizontal green
        signal 3 = horizontal yellow
    */

    bool verticalGreen = (signal == 0);
    bool horizontalGreen = (signal == 2);


    /*
        Stopping positions

        These values stop the complete realistic car
        just before the zebra crossing.
    */

    const float stopNS = 52.0;
    const float stopSN = -74.0;

    const float stopEW = -74.0;
    const float stopWE = 52.0;


    /*
        All cars move every frame.

        Red/yellow-side cars approach their stopping
        positions instead of remaining outside.
    */

    // North to South, moving downward
    moveNegativeCar(
        carNS1,
        stopNS,
        verticalGreen
    );

    moveNegativeCar(
        carNS2,
        stopNS,
        verticalGreen
    );


    // South to North, moving upward
    movePositiveCar(
        carSN1,
        stopSN,
        verticalGreen
    );

    movePositiveCar(
        carSN2,
        stopSN,
        verticalGreen
    );


    // West to East, moving right
    movePositiveCar(
        carEW1,
        stopEW,
        horizontalGreen
    );

    movePositiveCar(
        carEW2,
        stopEW,
        horizontalGreen
    );


    // East to West, moving left
    moveNegativeCar(
        carWE1,
        stopWE,
        horizontalGreen
    );

    moveNegativeCar(
        carWE2,
        stopWE,
        horizontalGreen
    );


    /*
        Smooth looping

        Each car is reset only after its complete
        body has left the visible window.
    */

    // North-to-South cars
    if (carNS1 < -123)
    {
        carNS1 = 130;
    }

    if (carNS2 < -123)
    {
        carNS2 = 155;
    }


    // South-to-North cars
    if (carSN1 > 103)
    {
        carSN1 = -130;
    }

    if (carSN2 > 103)
    {
        carSN2 = -155;
    }


    // West-to-East cars
    if (carEW1 > 103)
    {
        carEW1 = -130;
    }

    if (carEW2 > 103)
    {
        carEW2 = -155;
    }


    // East-to-West cars
    if (carWE1 < -123)
    {
        carWE1 = 130;
    }

    if (carWE2 < -123)
    {
        carWE2 = 155;
    }


    /*
        Draw vertical cars
    */

    // North to South
    carModel(
        -20,
        carNS1,
        0.05, 0.30, 0.95,
        0
    );

    carModel(
        -8,
        carNS2,
        0.90, 0.08, 0.08,
        0
    );


    // South to North
    carModel(
        10,
        carSN1,
        0.05, 0.78, 0.25,
        1
    );

    carModel(
        22,
        carSN2,
        0.95, 0.72, 0.05,
        1
    );


    /*
        Draw horizontal cars
    */

    // West to East
    carModel(
        carEW1,
        23,
        0.95, 0.28, 0.04,
        2
    );

    carModel(
        carEW2,
        7,
        0.05, 0.68, 0.90,
        2
    );


    // East to West
    carModel(
        carWE1,
        -9,
        0.62, 0.15, 0.72,
        3
    );

    carModel(
        carWE2,
        -25,
        0.95, 0.42, 0.04,
        3
    );
}

void transparentRect(float x1, float y1,
                     float x2, float y2,
                     float r, float g, float b,
                     float alpha)
{
    glColor4f(r, g, b, alpha);

    glBegin(GL_QUADS);

    glVertex2f(x1, y1);
    glVertex2f(x2, y1);
    glVertex2f(x2, y2);
    glVertex2f(x1, y2);

    glEnd();
}


void transparentText(float x, float y,
                     const char str[],
                     float r, float g, float b,
                     float alpha)
{
    glColor4f(r, g, b, alpha);
    glRasterPos2f(x, y);

    for (int i = 0; str[i] != '\0'; i++)
    {
        glutBitmapCharacter(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }
}


void fadedText(float x, float y,
               const char str[],
               float r, float g, float b)
{
    // Soft blurry shadow

        0.12

    transparentText(

        str,
        0, 0, 0,
        0.12
    );

    // Main faded writing
    transparentText(
        x, y,
        str,
        r, g, b,
        0.48
    );
}

void centerRoadDesign()
{
    float left = -30;
    float right = 30;
    float bottom = -30;
    float top = 30;

    glColor3f(0.55, 0.55, 0.55);
    glLineWidth(2.0);

    // Outer square border
    glBegin(GL_LINE_LOOP);

    glVertex2f(left, bottom);
    glVertex2f(right, bottom);
    glVertex2f(right, top);
    glVertex2f(left, top);

    glEnd();

    // Diagonal box-junction lines
    glBegin(GL_LINES);

    for (float c = -45; c <= 45; c += 20)
    {
        float x1;
        float y1;
        float x2;
        float y2;

        // First diagonal direction: x + y = c
        x1 = left;
        y1 = c - x1;

        if (y1 < bottom)
        {
            y1 = bottom;
            x1 = c - y1;
        }
        else if (y1 > top)
        {
            y1 = top;
            x1 = c - y1;
        }

        x2 = right;
        y2 = c - x2;

        if (y2 < bottom)
        {
            y2 = bottom;
            x2 = c - y2;
        }
        else if (y2 > top)
        {
            y2 = top;
            x2 = c - y2;
        }

        glVertex2f(x1, y1);
        glVertex2f(x2, y2);

        // Second diagonal direction: x - y = c
        x1 = left;
        y1 = x1 - c;

        if (y1 < bottom)
        {
            y1 = bottom;
            x1 = y1 + c;
        }
        else if (y1 > top)
        {
            y1 = top;
            x1 = y1 + c;
        }

        x2 = right;
        y2 = x2 - c;

        if (y2 < bottom)
        {
            y2 = bottom;
            x2 = y2 + c;
        }
        else if (y2 > top)
        {
            y2 = top;
            x2 = y2 + c;
        }

        glVertex2f(x1, y1);
        glVertex2f(x2, y2);
    }

    glEnd();

    // Return line thickness to normal
    glLineWidth(1.0);
}
void weatherEffect()
{
    // Rain effect
    if (mode == 2)
    {
        glColor3f(0.5, 0.7, 1.0);

        glBegin(GL_LINES);

        for (int i = 0; i < 150; i++)
        {
            glVertex2f(
                rainX[i],
                rainY[i]
            );

            glVertex2f(
                rainX[i] - 2,
                rainY[i] - 8
            );

            rainY[i] -= 1.0;

            if (rainY[i] < -100)
            {
                rainY[i] = 100;
                rainX[i] = -100 + rand() % 200;
            }
        }

        glEnd();
    }

    // Snow effect
    else if (mode == 3)
    {
        for (int i = 0; i < 150; i++)
        {
            circle(
                snowX[i],
                snowY[i],
                1.5,
                1.0,
                1.0,
                1.0
            );

            snowY[i] -= 0.3;

            if (snowY[i] < -100)
            {
                snowY[i] = 100;
                snowX[i] = -100 + rand() % 200;
            }
        }
    }
}
void display()
{
    if (mode == 1)
    {
        glClearColor(0.1, 0.55, 0.9, 1);
    }

    if (mode == 2)
    {
        glClearColor(0.12, 0.15, 0.22, 1);
    }

    if (mode == 3)
    {
        glClearColor(0.75, 0.85, 0.95, 1);
    }

    glClear(GL_COLOR_BUFFER_BIT);

    sidewalk();
    road();
    centerRoadDesign();

    // Transparent keyboard-command guide
    controlPanel();

    environment();
    boards();


    vehicles();
    trafficSystem();

    weatherEffect();

    glFlush();
}

void animation(int)
{
    glutPostRedisplay();

    glutTimerFunc(
        30,
        animation,
        0
    );
}


void signalTimer(int)
{
    timerValue--;

    if (timerValue <= 0)
    {
        if (signal == 0)
        {
            signal = 1;
            timerValue = yellowTime;
        }

        else if (signal == 1)
        {
            signal = 2;
            timerValue = 40;
        }

        else if (signal == 2)
        {
            signal = 3;
            timerValue = yellowTime;
        }

        else if (signal == 3)
        {
            signal = 0;
            timerValue = 35;
        }
    }

    glutTimerFunc(
        1000,
        signalTimer,
        0
    );
}

void updateCarSpeed()
{
    float normalSpeed;

    // Normal speed according to weather mode
    if (mode == 1)
    {
        normalSpeed = 0.35;   // Sunny: fastest
    }
    else if (mode == 2)
    {
        normalSpeed = 0.22;   // Rain: medium
    }
    else
    {
        normalSpeed = 0.10;   // Snow: slowest
    }

    // Apply 2x speed when Space is activated
    if (doubleSpeed)
    {
        speedValue = normalSpeed * 2.0;
    }
    else
    {
        speedValue = normalSpeed;
    }
}

void keyboard(unsigned char key, int x, int y)
{
    // Mode 1: Sunny
    if (key == '1')
    {
        mode = 1;
        updateCarSpeed();
    }

    // Mode 2: Rain
    else if (key == '2')
    {
        mode = 2;
        updateCarSpeed();
    }

    // Mode 3: Snow
    else if (key == '3')
    {
        mode = 3;
        updateCarSpeed();
    }

    // Space key: toggle normal and 2x speed
    else if (key == ' ')
    {
        doubleSpeed = !doubleSpeed;
        updateCarSpeed();
    }

    // ESC key: close the program
    else if (key == 27)
    {
        exit(0);
    }

    glutPostRedisplay();
}


void init()
{
    glEnable(GL_BLEND);

    glBlendFunc(
        GL_SRC_ALPHA,
        GL_ONE_MINUS_SRC_ALPHA
    );

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();

    gluOrtho2D(-100, 100, -100, 100);

    for (int i = 0; i < 150; i++)
    {
        rainX[i] = -100 + rand() % 200;
        rainY[i] = -100 + rand() % 200;

        snowX[i] = -100 + rand() % 200;
        snowY[i] = -100 + rand() % 200;
    }
}


int main(int argc, char **argv)
{
    glutInit(&argc, argv);

    glutInitDisplayMode(
        GLUT_SINGLE | GLUT_RGB
    );

    glutInitWindowSize(
        1100,
        800
    );

    glutCreateWindow(
        "Traffic Control System"
    );

    init();

    glutDisplayFunc(display);
    glutKeyboardFunc(keyboard);

    glutTimerFunc(
        30,
        animation,
        0
    );

    glutTimerFunc(
        1000,
        signalTimer,
        0
    );

    glutMainLoop();

    return 0;
}
