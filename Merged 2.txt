#include <windows.h>
#include <GL/glut.h>
#include <math.h>


void rect(float x1,float y1,float x2,float y2,
          float r,float g,float b)
{
    glColor3f(r,g,b);

    glBegin(GL_POLYGON);

    glVertex2f(x1,y1);
    glVertex2f(x2,y1);
    glVertex2f(x2,y2);
    glVertex2f(x1,y2);

    glEnd();
}


void circle(float x,float y,float radius,
            float r,float g,float b)
{
    glColor3f(r,g,b);

    glBegin(GL_POLYGON);

    for(int i=0;i<100;i++)
    {
        float angle=2*3.1416*i/100;

        glVertex2f(
            x+radius*cos(angle),
            y+radius*sin(angle)
        );
    }

    glEnd();
}


void text(float x,float y,const char str[],
          float r,float g,float b)
{
    glColor3f(r,g,b);

    glRasterPos2f(x,y);

    for(int i=0;str[i]!='\0';i++)
    {
        glutBitmapCharacter(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }
}


void sidewalk()
{
    float r=0.35f;
    float g=0.35f;
    float b=0.35f;


    rect(-100,100,-40,40,
         r,g,b);

    rect(40,100,100,40,
         r,g,b);


    rect(-100,-40,-40,-100,
         r,g,b);

    rect(40,-40,100,-100,
         r,g,b);
}



void road()
{
    float roadR=0.06f;
    float roadG=0.06f;
    float roadB=0.07f;


    rect(-100,35,100,-35,
         roadR,roadG,roadB);


    rect(-35,100,35,-100,
         roadR,roadG,roadB);



    rect(-100,40,100,35,
         0.55,0.55,0.55);


    rect(-100,-35,100,-40,
         0.55,0.55,0.55);



    rect(-40,100,-35,-100,
         0.55,0.55,0.55);


    rect(35,100,40,-100,
         0.55,0.55,0.55);



    rect(-3,100,3,52,
         1,0.8,0);


    rect(-3,-52,3,-100,
         1,0.8,0);



    rect(-100,-3,-52,3,
         1,0.8,0);


    rect(52,-3,100,3,
         1,0.8,0);
}


void laneMarking()
{
    for(float y=58;y<=90;y+=16)
    {
        rect(-28,y,-22,y+8,
             1,1,1);

        rect(22,y,28,y+8,
             1,1,1);
    }


    for(float y=-98;y<=-58;y+=16)
    {
        rect(-28,y,-22,y+8,
             1,1,1);

        rect(22,y,28,y+8,
             1,1,1);
    }


    for(float x=-98;x<=-58;x+=16)
    {
        rect(x,-28,x+8,-22,
             1,1,1);

        rect(x,22,x+8,28,
             1,1,1);
    }


    for(float x=58;x<=90;x+=16)
    {
        rect(x,-28,x+8,-22,
             1,1,1);

        rect(x,22,x+8,28,
             1,1,1);
    }
}


void zebraCrossing()
{
    for(int i=0;i<10;i++)
    {
        float position=-30+i*6;


        rect(position,40,position+3.2,50,
             1,1,1);


        rect(position,-50,position+3.2,-40,
             1,1,1);


        rect(40,position,50,position+3.2,
             1,1,1);


        rect(-50,position,-40,position+3.2,
             1,1,1);
    }
}


void tree(float x,float y)
{
    circle(x,y,5,
           0.22,0.22,0.22);


    rect(x-2,y,x+2,y+13,
         0.38,0.20,0.07);


    circle(x,y+18,7,
           0.04,0.42,0.08);


    circle(x-5,y+15,5.5,
           0.06,0.55,0.10);


    circle(x+5,y+15,5.5,
           0.06,0.55,0.10);


    circle(x,y+22,5,
           0.06,0.60,0.13);
}void messageBoard(float x,float y)

        x+1,

    );


    rect(
        x+1.5,

        x+33.5,

        0.28,
        0.36,
        0.40
    );


    rect(

        x+5,

void display()
{
    glClearColor(0.1,0.55,0.9,1);

    glClear(GL_COLOR_BUFFER_BIT);


    sidewalk();

    road();

    laneMarking();

    zebraCrossing();


    tree(-50,-96);


    messageBoard(-98,70);
    messageBoard(63,70);
    messageBoard(-98,-85);
    messageBoard(63,-85);


    glFlush();
}



void init()
{
    glMatrixMode(GL_PROJECTION);

    glLoadIdentity();

    gluOrtho2D(-100,100,-100,100);
}



int main(int argc,char **argv)
{
    glutInit(&argc,argv);


    glutInitDisplayMode(
        GLUT_SINGLE |
        GLUT_RGB
    );


    glutInitWindowSize(
        1100,
        800
    );


    glutCreateWindow(
        "Road"
    );


    init();


    glutDisplayFunc(display);


    glutMainLoop();


    return 0;
}
