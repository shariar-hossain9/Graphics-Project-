// ============================================================================
// Function: rect()
// Purpose : Draws a filled rectangle using four corner coordinates and an RGB color.
// ============================================================================
void rect(float x1, float y1, float x2, float y2,
          // Declares a variable used by this part of the program.
          float r, float g, float b)
{
    // Sets the current drawing color using red, green, and blue values from 0.0 to 1.0.
    glColor3f(r, g, b);

    // Begins drawing one or more four-sided filled shapes.
    glBegin(GL_QUADS);

    // Adds a 2D vertex to the shape currently being drawn.
    glVertex2f(x1, y1);
    // Adds a 2D vertex to the shape currently being drawn.
    glVertex2f(x2, y1);
    // Adds a 2D vertex to the shape currently being drawn.
    glVertex2f(x2, y2);
    // Adds a 2D vertex to the shape currently being drawn.
    glVertex2f(x1, y2);

    // Finishes the current OpenGL drawing block.
    glEnd();
}


// ============================================================================
// Function: circle()
// Purpose : Draws a filled circle by generating 100 points around its circumference.
// ============================================================================
void circle(float x, float y, float radius,
            // Declares a variable used by this part of the program.
            float red, float green, float blue)
{
    // Sets the current drawing color using red, green, and blue values from 0.0 to 1.0.
    glColor3f(red, green, blue);

    // Begins drawing a filled polygon.
    glBegin(GL_POLYGON);

    // Starts a loop that repeats the following block for a fixed range of values.
    for (int i = 0; i < 100; i++)
    {
        // Declares a variable used by this part of the program.
        float angle = 2 * 3.1416 * i / 100;

        // Adds a 2D vertex to the shape currently being drawn.
        glVertex2f(
            x + radius * cos(angle),
            y + radius * sin(angle)
        );
    }

    // Finishes the current OpenGL drawing block.
    glEnd();
}


// ============================================================================
// Function: ellipse()
// Purpose : Draws a transparent or solid ellipse using separate horizontal and vertical radii.
// ============================================================================
void ellipse(float x, float y,
             // Declares a variable used by this part of the program.
             float radiusX, float radiusY,
             // Declares a variable used by this part of the program.
             float red, float green, float blue,
             // Declares a variable used by this part of the program.
             float alpha)
{
    // Sets the drawing color and alpha transparency.
    glColor4f(red, green, blue, alpha);

    // Begins drawing a filled polygon.
    glBegin(GL_POLYGON);

    // Starts a loop that repeats the following block for a fixed range of values.
    for (int i = 0; i < 100; i++)
    {
        // Declares a variable used by this part of the program.
        float angle = 2.0f * 3.1416f * i / 100.0f;

        // Adds a 2D vertex to the shape currently being drawn.
        glVertex2f(
            x + radiusX * cos(angle),
            y + radiusY * sin(angle)
        );
    }

    // Finishes the current OpenGL drawing block.
    glEnd();
}



// ============================================================================
// Function: text()
// Purpose : Displays bitmap text at a specified OpenGL coordinate.
// ============================================================================
void text(float x, float y, const char str[],
          // Declares a variable used by this part of the program.
          float r, float g, float b)
{
    // Sets the current drawing color using red, green, and blue values from 0.0 to 1.0.
    glColor3f(r, g, b);
    // Sets the screen position where bitmap text will begin.
    glRasterPos2f(x, y);

    // Starts a loop that repeats the following block for a fixed range of values.
    for (int i = 0; str[i] != '\0'; i++)
    {
        // Draws one character using the selected GLUT bitmap font.
        glutBitmapCharacter(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }
}

// ============================================================================
// Function: centeredText()
// Purpose : Calculates the width of a message and draws it horizontally centered around a given coordinate.
// ============================================================================
void centeredText(float centerX, float y,
                  const char str[],
                  // Declares a variable used by this part of the program.
                  float r, float g, float b)
{
    // Declares a variable used by this part of the program.
    int pixelWidth = 0;

    // Starts a loop that repeats the following block for a fixed range of values.
    for (int i = 0; str[i] != '\0'; i++)
    {
        pixelWidth += glutBitmapWidth(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }

    // Declares a variable used by this part of the program.
    float windowWidth = glutGet(GLUT_WINDOW_WIDTH);
    // Declares a variable used by this part of the program.
    float worldWidth = pixelWidth * 200.0f / windowWidth;

    // Calls text() to draw or update this specific object.
    text(
        centerX - worldWidth / 2.0f,
        y,
        str,
        r, g, b
    );
}

                     bool greenSignal)

            signal = 3;
            // Assigns or updates a value used in the current calculation.
            timerValue = yellowTime;
        }
        // Runs this block when the previous conditions were false.
        else
        {
            // Changes the current traffic-light phase.
            signal = 0;
            // Assigns or updates a value used in the current calculation.
            timerValue = 35;
        }
    }

    // Registers a timer callback that GLUT will call after the specified delay.
    glutTimerFunc(
        1000,
        signalTimer,
        0
    );
}


// ============================================================================
// Function: updateCarSpeed()
// Purpose : Selects vehicle speed from the current weather mode and applies the optional double-speed setting.
// ============================================================================
void updateCarSpeed()
{
    // Declares a variable used by this part of the program.
    float normalSpeed;

    // Checks this condition and executes the following block only when it is true.
    if (mode == 1)
    {
        // Assigns or updates a value used in the current calculation.
        normalSpeed = 0.35;
    }
    // Checks another condition when the previous condition was false.
    else if (mode == 2)
    {
        // Assigns or updates a value used in the current calculation.
        normalSpeed = 0.22;
    }
    // Runs this block when the previous conditions were false.
    else
    {
        // Assigns or updates a value used in the current calculation.
        normalSpeed = 0.10;
    }

    // Checks this condition and executes the following block only when it is true.
    if (doubleSpeed)
    {
        // Stores the speed that will be used by all moving cars.
        speedValue = normalSpeed * 2.0;
    }
    // Runs this block when the previous conditions were false.
    else
    {
        // Stores the speed that will be used by all moving cars.
        speedValue = normalSpeed;
    }
}


// ============================================================================
// Function: keyboard()
// Purpose : Handles keyboard input for weather selection, double speed, and program exit.
// ============================================================================
void keyboard(unsigned char key, int x, int y)
{
    // Checks this condition and executes the following block only when it is true.
    if (key == '1')
    {
        // Changes the active weather mode.
        mode = 1;
        updateCarSpeed();
    }
    // Checks another condition when the previous condition was false.
    else if (key == '2')
    {
        // Changes the active weather mode.
        mode = 2;
        updateCarSpeed();
    }
    // Checks another condition when the previous condition was false.
    else if (key == '3')
    {
        // Changes the active weather mode.
        mode = 3;
        updateCarSpeed();
    }
    // Checks another condition when the previous condition was false.
    else if (key == ' ')
    {
        // Toggles double speed: true becomes false, and false becomes true.
        doubleSpeed = !doubleSpeed;
        updateCarSpeed();
    }
    // Checks another condition when the previous condition was false.
    else if (key == 27)
    {
        // Closes the program normally when Escape is pressed.
        exit(0);
    }

    // Marks the window for redrawing during the next GLUT event cycle.
    glutPostRedisplay();
}


// ============================================================================
// Function: init()
// Purpose : Initializes blending, the 2D coordinate system, and random rain/snow particle positions.
// ============================================================================
void init()
{
    // Enables alpha blending so transparent rain, snow, and reflection effects can be displayed.
    glEnable(GL_BLEND);

    // Defines how transparent source colors are combined with colors already on the screen.
    glBlendFunc(
        GL_SRC_ALPHA,
        GL_ONE_MINUS_SRC_ALPHA
    );

    // Selects the projection matrix for setting the 2D viewing area.
    glMatrixMode(GL_PROJECTION);
    // Resets the selected matrix to its default identity state.
    glLoadIdentity();

    // Creates a 2D coordinate system ranging from -100 to 100 on both axes.
    gluOrtho2D(-100, 100, -100, 100);

    // Starts a loop that repeats the following block for a fixed range of values.
    for (int i = 0; i < 150; i++)
    {
        // Generates a random coordinate within the visible -100 to 99 range.
        rainX[i] = -100 + rand() % 200;
        // Generates a random coordinate within the visible -100 to 99 range.
        rainY[i] = -100 + rand() % 200;

        // Generates a random coordinate within the visible -100 to 99 range.
        snowX[i] = -100 + rand() % 200;
        // Generates a random coordinate within the visible -100 to 99 range.
        snowY[i] = -100 + rand() % 200;
    }
}


// ============================================================================
// Function: main()
// Purpose : Program entry point. It creates the GLUT window, registers callbacks, starts timers, and enters the event loop.
// ============================================================================
// Declares a variable used by this part of the program.
int main(int argc, char **argv)
{
    // Initializes the GLUT library using the program's command-line arguments.
    glutInit(&argc, argv);

    // Uses a single-buffered RGB window.
    glutInitDisplayMode(
        GLUT_SINGLE | GLUT_RGB
    );

    // Sets the initial width and height of the program window.
    glutInitWindowSize(
        1100,
        800
    );

    // Creates the OpenGL window and assigns its title.
    glutCreateWindow(
        "Traffic Control System"
    );

    init();

    // Registers display() as the function responsible for drawing each frame.
    glutDisplayFunc(display);
    // Registers keyboard() as the keyboard-input callback.
    glutKeyboardFunc(keyboard);

    // Registers a timer callback that GLUT will call after the specified delay.
    glutTimerFunc(
        30,
        animation,
        0
    );

    // Registers a timer callback that GLUT will call after the specified delay.
    glutTimerFunc(
        1000,
        signalTimer,
        0
    );

    // Starts GLUT's event loop. Control normally remains here until the program exits.
    glutMainLoop();

    // Ends main() and reports successful program termination.
    return 0;
}
