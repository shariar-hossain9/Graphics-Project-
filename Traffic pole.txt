#include <windows.h>
#include <GL/glut.h>
#include <math.h>
#include <stdlib.h>
#include <stdio.h>


int timerValue = 31;


void rect(float x1, float y1, float x2, float y2,
          float r, float g, float b)
{
    glColor3f(r,g,b);

    glBegin(GL_QUADS);

    glVertex2f(x1,y1);
    glVertex2f(x2,y1);
    glVertex2f(x2,y2);
    glVertex2f(x1,y2);

    glEnd();
}



void circle(float x,float y,float radius,
            float red,float green,float blue)
{
    glColor3f(red,green,blue);

    glBegin(GL_POLYGON);

    for(int i=0;i<100;i++)
    {
        float angle = 2 * 3.1416 * i / 100;

        glVertex2f(
            x + radius*cos(angle),
            y + radius*sin(angle)
        );
    }

    glEnd();
}



void text(float x,float y,const char str[],
          float r,float g,float b)
{
    glColor3f(r,g,b);

    glRasterPos2f(x,y);

    for(int i=0;str[i]!='\0';i++)
    {
        glutBitmapCharacter(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }
}
void trafficPole(float x, float y, int active)
{
    int side = (x < 0) ? 1 : -1;

    float lightX = x + side * 14;
    float timerX = x + side * 5;

    float armLeft;
    float armRight;


    if (side == 1)
    {
        armLeft = x;
        armRight = lightX;
    }
    else
    {
        armLeft = lightX;
        armRight = x;
    }


    rect(
        x - 1.7,
        y - 8,
        x + 1.7,
        y + 29,
        0.07,
        0.10,
        0.13
    );


    rect(
        x - 1.1,
        y - 7,
        x + 1.1,
        y + 28,
        0.28,
        0.38,
        0.45
    );


    rect(
        x - 0.35,
        y - 6,
        x + 0.15,
        y + 27,
        0.65,
        0.72,
        0.76
    );


    rect(
        x - 4.5,
        y - 10,
        x + 4.5,
        y - 7,
        0.04,
        0.06,
        0.08
    );


    rect(
        x - 3.6,
        y - 9.2,
        x + 3.6,
        y - 7,
        0.28,
        0.38,
        0.45
    );


    rect(
        armLeft,
        y + 26,
        armRight,
        y + 29,
        0.07,
        0.10,
        0.13
    );


    rect(
        armLeft,
        y + 26.6,
        armRight,
        y + 28.4,
        0.28,
        0.38,
        0.45
    );


    rect(
        armLeft + 0.8,
        y + 27.6,
        armRight - 0.8,
        y + 28.1,
        0.65,
        0.72,
        0.76
    );


    rect(
        lightX - 4.8,
        y + 9,
        lightX + 4.8,
        y + 32,
        0.03,
        0.04,
        0.05
    );


    rect(
        lightX - 4,
        y + 10,
        lightX + 4,
        y + 31,
        0.09,
        0.11,
        0.12
    );


    circle(
        lightX,
        y + 27,
        2.7,
        active == 0 ? 1.0 : 0.18,
        0,
        0
    );


    circle(
        lightX,
        y + 20.5,
        2.7,
        active == 1 ? 1.0 : 0.18,
        active == 1 ? 0.78 : 0.08,
        0
    );


    circle(
        lightX,
        y + 14,
        2.7,
        0,
        active == 2 ? 1.0 : 0.18,
        0
    );
}
void display()
{
    glClearColor(
        1,
        1,
        1,
        1
    );


    glClear(GL_COLOR_BUFFER_BIT);


    trafficPole(
        -20,
        -30,
        0
    );


    glFlush();
}



void init()
{
    glMatrixMode(GL_PROJECTION);

    glLoadIdentity();


    gluOrtho2D(
        -100,
        100,
        -100,
        100
    );
}



int main(int argc,char **argv)
{
    glutInit(&argc,argv);


    glutInitDisplayMode(
        GLUT_SINGLE |
        GLUT_RGB
    );


    glutInitWindowSize(
        800,
        800
    );


    glutCreateWindow(
        "Traffic Pole"
    );


    init();


    glutDisplayFunc(display);


    glutMainLoop();


    return 0;
}
