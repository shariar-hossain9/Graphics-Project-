void rect(float x1, float y1, float x2, float y2,
          float r, float g, float b)
{
    glColor3f(r, g, b);

    glBegin(GL_QUADS);

    glVertex2f(x1, y1);
    glVertex2f(x2, y1);
    glVertex2f(x2, y2);
    glVertex2f(x1, y2);

    glEnd();
}


void circle(float x, float y, float radius,
            float red, float green, float blue)
{
    glColor3f(red, green, blue);

    glBegin(GL_POLYGON);

    for (int i = 0; i < 100; i++)
    {
        float angle = 2 * 3.1416 * i / 100;

        glVertex2f(
            x + radius * cos(angle),
            y + radius * sin(angle)
        );
    }

    glEnd();
}


void text(float x, float y, const char str[],
          float r, float g, float b)
{
    glColor3f(r, g, b);
    glRasterPos2f(x, y);

    for (int i = 0; str[i] != '\0'; i++)
    {
        glutBitmapCharacter(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }
}

void centeredText(float centerX, float y,
                  const char str[],
                  float r, float g, float b)
{
    int pixelWidth = 0;


    for (int i = 0; str[i] != '\0'; i++)
    {
        pixelWidth += glutBitmapWidth(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }


    float windowWidth = glutGet(GLUT_WINDOW_WIDTH);
    float worldWidth = pixelWidth * 200.0f / windowWidth;


    text(
        centerX - worldWidth / 2.0f,
        y,
        str,
        r, g, b
    );
}

bool checkVerticalCollision(float front, float back)
{
    if (fabs(front - back) < safeDistance)
    {
        return true;
    }

    return false;
}


bool checkHorizontalCollision(float front, float back)
{
    if (fabs(front - back) < safeDistance)
    {
        return true;
    }

    return false;
}

void moveNegativeCar(float &position,
                     float stopPosition,
                     bool greenSignal)
{



    if (greenSignal || position < stopPosition)
    {
        position -= speedValue;
    }
    else if (position > stopPosition)
    {

        position -= speedValue;


        if (position < stopPosition)
        {
            position = stopPosition;
        }
    }
}


void movePositiveCar(float &position,
                     float stopPosition,
                     bool greenSignal)
{



    if (greenSignal || position > stopPosition)
    {
        position += speedValue;
    }
    else if (position < stopPosition)
    {

        position += speedValue;


        if (position > stopPosition)
        {
            position = stopPosition;
        }
    }
}

void transparentRect(float x1, float y1,
                     float x2, float y2,
                     float r, float g, float b,
                     float alpha)
{
    glColor4f(r, g, b, alpha);

    glBegin(GL_QUADS);

    glVertex2f(x1, y1);
    glVertex2f(x2, y1);
    glVertex2f(x2, y2);
    glVertex2f(x1, y2);

    glEnd();
}


void transparentText(float x, float y,
                     const char str[],
                     float r, float g, float b,
                     float alpha)
{
    glColor4f(r, g, b, alpha);
    glRasterPos2f(x, y);

    for (int i = 0; str[i] != '\0'; i++)
    {
        glutBitmapCharacter(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }
}

void centerRoadDesign()
{
    float left = -30;
    float right = 30;
    float bottom = -30;
    float top = 30;

    glColor3f(0.55, 0.55, 0.55);
    glLineWidth(2.0);


    glBegin(GL_LINE_LOOP);

    glVertex2f(left, bottom);
    glVertex2f(right, bottom);
    glVertex2f(right, top);
    glVertex2f(left, top);

    glEnd();


    glBegin(GL_LINES);

    for (float c = -45; c <= 45; c += 20)
    {
        float x1;
        float y1;
        float x2;
        float y2;


        x1 = left;
        y1 = c - x1;

        if (y1 < bottom)
        {
            y1 = bottom;
            x1 = c - y1;
        }
        else if (y1 > top)
        {
            y1 = top;
            x1 = c - y1;
        }

        x2 = right;
        y2 = c - x2;

        if (y2 < bottom)
        {
            y2 = bottom;
            x2 = c - y2;
        }
        else if (y2 > top)
        {
            y2 = top;
            x2 = c - y2;
        }

        glVertex2f(x1, y1);
        glVertex2f(x2, y2);


        x1 = left;
        y1 = x1 - c;

        if (y1 < bottom)
        {
            y1 = bottom;
            x1 = y1 + c;
        }
        else if (y1 > top)
        {
            y1 = top;
            x1 = y1 + c;
        }

        x2 = right;
        y2 = x2 - c;

        if (y2 < bottom)
        {
            y2 = bottom;
            x2 = y2 + c;
        }
        else if (y2 > top)
        {
            y2 = top;
            x2 = y2 + c;
        }

        glVertex2f(x1, y1);
        glVertex2f(x2, y2);
    }

    glEnd();


    glLineWidth(1.0);
}
void weatherEffect()
{

    if (mode == 2)
    {
        glColor3f(0.5, 0.7, 1.0);

        glBegin(GL_LINES);

        for (int i = 0; i < 150; i++)
        {
            glVertex2f(
                rainX[i],
                rainY[i]
            );

            glVertex2f(
                rainX[i] - 2,
                rainY[i] - 8
            );

            rainY[i] -= 1.0;

            if (rainY[i] < -100)
            {
                rainY[i] = 100;
                rainX[i] = -100 + rand() % 200;
            }
        }

        glEnd();
    }


    else if (mode == 3)
    {
        for (int i = 0; i < 150; i++)
        {
            circle(
                snowX[i],
                snowY[i],
                1.5,
                1.0,
                1.0,
                1.0
            );

            snowY[i] -= 0.3;

            if (snowY[i] < -100)
            {
                snowY[i] = 100;
                snowX[i] = -100 + rand() % 200;
            }
        }
    }
}

{

    }

void animation(int)
{
    glutPostRedisplay();

    glutTimerFunc(
        30,
        animation,
        0
    );
}


void signalTimer(int)
{
    timerValue--;

    if (timerValue <= 0)
    {
        if (signal == 0)
        {
            signal = 1;
            timerValue = yellowTime;
        }

        else if (signal == 1)
        {
            signal = 2;
            timerValue = 40;
        }

        else if (signal == 2)
        {
            signal = 3;
            timerValue = yellowTime;
        }

        else if (signal == 3)
        {
            signal = 0;
            timerValue = 35;
        }
    }

    glutTimerFunc(
        1000,
        signalTimer,
        0
    );
}

void updateCarSpeed()
{
    float normalSpeed;


    if (mode == 1)
    {
        normalSpeed = 0.35;
    }
    else if (mode == 2)
    {
        normalSpeed = 0.22;
    }
    else
    {
        normalSpeed = 0.10;
    }


    if (doubleSpeed)
    {
        speedValue = normalSpeed * 2.0;
    }
    else
    {
        speedValue = normalSpeed;
    }
}

void keyboard(unsigned char key, int x, int y)
{

    if (key == '1')
    {
        mode = 1;
        updateCarSpeed();
    }


    else if (key == '2')
    {
        mode = 2;
        updateCarSpeed();
    }


    else if (key == '3')
    {
        mode = 3;
        updateCarSpeed();
    }


    else if (key == ' ')
    {
        doubleSpeed = !doubleSpeed;
        updateCarSpeed();
    }


    else if (key == 27)
    {
        exit(0);
    }

    glutPostRedisplay();
}


void init()
{
    glEnable(GL_BLEND);

    glBlendFunc(
        GL_SRC_ALPHA,
        GL_ONE_MINUS_SRC_ALPHA
    );

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();

    gluOrtho2D(-100, 100, -100, 100);

    for (int i = 0; i < 150; i++)
    {
        rainX[i] = -100 + rand() % 200;
        rainY[i] = -100 + rand() % 200;

        snowX[i] = -100 + rand() % 200;
        snowY[i] = -100 + rand() % 200;
    }
}


int main(int argc, char **argv)
{
    glutInit(&argc, argv);

    glutInitDisplayMode(
        GLUT_SINGLE | GLUT_RGB
    );

    glutInitWindowSize(
        1100,
        800
    );

    glutCreateWindow(
        "Traffic Control System"
    );

    init();

    glutDisplayFunc(display);
    glutKeyboardFunc(keyboard);

    glutTimerFunc(
        30,
        animation,
        0
    );

    glutTimerFunc(
        1000,
        signalTimer,
        0
    );

    glutMainLoop();

    return 0;
}
