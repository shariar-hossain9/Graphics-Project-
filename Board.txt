#include <windows.h>
#include <GL/glut.h>
#include <math.h>


void rect(float x1,float y1,float x2,float y2,
          float r,float g,float b)
{
    glColor3f(r,g,b);

    glBegin(GL_QUADS);

    glVertex2f(x1,y1);
    glVertex2f(x2,y1);
    glVertex2f(x2,y2);
    glVertex2f(x1,y2);

    glEnd();
}



void circle(float x,float y,float radius,
            float r,float g,float b)
{
    glColor3f(r,g,b);

    glBegin(GL_POLYGON);

    for(int i=0;i<100;i++)
    {
        float angle=2*3.1416*i/100;

        glVertex2f(
            x+radius*cos(angle),
            y+radius*sin(angle)
        );
    }

    glEnd();
}



void text(float x,float y,const char str[],
          float r,float g,float b)
{
    glColor3f(r,g,b);

    glRasterPos2f(x,y);

    for(int i=0;str[i]!='\0';i++)
    {
        glutBitmapCharacter(
            GLUT_BITMAP_HELVETICA_18,
            str[i]
        );
    }
}



void messageBoard(float x,float y)
{

    rect(
        x+1.2,
        y-1.2,
        x+36.2,
        y+18.8,
        0.03,
        0.04,
        0.05
    );


    rect(
        x,
        y,
        x+35,
        y+20,
        0.10,
        0.16,
        0.20
    );


    rect(
        x+1,
        y+18.5,
        x+34,
        y+19.5,
        0.42,
        0.52,
        0.57
    );


    rect(
        x+1.5,
        y+1.5,
        x+33.5,
        y+18.5,
        0.28,
        0.36,
        0.40
    );


    rect(
        x+3,
        y+3,
        x+32,
        y+17,
        0.03,
        0.28,
        0.12
    );


    circle(
        x+5,
        y+5,
        1.2,
        0.20,
        1.0,
        0.35
    );


    circle(
        x+30,
        y+5,
        1.2,
        0.20,
        1.0,
        0.35
    );


    text(
        x+9,
        y+9,
        "A Good Day",
        1.0,
        0.90,
        0.25
    );

}



void display()
{
    glClearColor(
        1.0,
        1.0,
        1.0,
        1
    );


    glClear(GL_COLOR_BUFFER_BIT);


    messageBoard(-18,-10);


    glFlush();
}



void init()
{
    glMatrixMode(GL_PROJECTION);

    glLoadIdentity();

    gluOrtho2D(
        -100,
        100,
        -100,
        100
    );
}



int main(int argc,char **argv)
{

    glutInit(&argc,argv);


    glutInitDisplayMode(
        GLUT_SINGLE |
        GLUT_RGB
    );


    glutInitWindowSize(
        800,
        800
    );


    glutCreateWindow(
        "Message Board Object"
    );


    init();


    glutDisplayFunc(display);


    glutMainLoop();


    return 0;
}
